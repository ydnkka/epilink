\tikzset{
  connector/.style={draw=BaseInk, line width=0.75pt},
  flow/.style={
    connector,
    -{Latex[length=2.1mm,width=1.4mm]},
    shorten >= 1pt
  },
  scenario flow/.style={
    -{Latex[length=1.5mm,width=1.0mm]},
    draw=BaseInk,
    line width=0.65pt,
    shorten >= 1pt,
    shorten <= 1pt
  },
  workflow box/.style={
    rounded corners=3pt,
    line width=0.75pt,
    rectangle split,
    rectangle split parts=2,
    font=\scriptsize,
    inner sep=4pt,
    align=left,
    text=BaseInk
  },
  neutral box/.style={
    workflow box,
    draw=NeutralGrey,
    rectangle split part fill={white, BoxBg}
  },
  shared box/.style={
    workflow box,
    draw=SharedColor,
    rectangle split part fill={SharedColor!12, white}
  },
  temporal box/.style={
    workflow box,
    draw=TemporalColor,
    rectangle split part fill={TemporalColor!12, white}
  },
  genetic box/.style={
    workflow box,
    draw=GeneticColor,
    rectangle split part fill={GeneticColor!12, white}
  },
  final box/.style={
    workflow box,
    draw=ScenarioColor,
    rectangle split part fill={ScenarioColor!15, white}
  },
  scenario box/.style={
    workflow box,
    draw=ScenarioColor,
    rectangle split part fill={ScenarioColor!12, white}
  },
  scenario column/.style={
    minimum width=5.0cm,
    minimum height=3.15cm,
    inner sep=0pt,
    anchor=south
  },
  sampled case/.style={
    circle,
    draw=BaseInk,
    fill=white,
    line width=0.75pt,
    minimum size=4mm,
    inner sep=0,
    font=\scriptsize\bfseries
  },
  latent case/.style={
    sampled case,
    fill=NeutralGrey!15,
    dash pattern=on 1.5pt off 1pt
  }
}

\newcommand{\boxheader}[1]{{\centering\bfseries #1\par}}

\begin{document}
\begin{tikzpicture}[font=\sffamily, x=1cm, y=1cm]

  % --- INPUT ---
  \node[neutral box, text width=3.85cm] (input) at (6.25, 13.0) {%
    \boxheader{Observed Pairwise Data}
    \nodepart{second}
    Testing-time difference: $t_{ij}$ \\
    Consensus genetic distance: $g_{ij}$
  };

  % --- TRANSMISSION SCENARIOS ---
  \node[scenario box, text width=10.5cm, anchor=north]
    (scenarios_border) at ($(input.south)+(0,-0.55)$) {
    \boxheader{Unobserved Transmission Scenarios ($\mathrm{S}_M$)}
    \nodepart{second}
    \rule{0pt}{3.15cm}
  };

  \node[scenario column] (ad_col)
    at ($(scenarios_border.south)+(-2.55,0.14)$) {};
  \node[anchor=north, font=\bfseries\tiny, text=BaseInk, yshift=-4pt]
    at (ad_col.north) {Ancestor-Descendant $H_{\mathrm{AD}}(m)$};
  \node[anchor=south, font=\tiny, text=BaseInk, yshift=3pt]
    at (ad_col.south) {Direct transmission $i \to j$, $m=0$};

  \node[scenario column] (ca_col)
    at ($(scenarios_border.south)+(2.55,0.14)$) {};
  \node[anchor=north, font=\bfseries\tiny, text=BaseInk, yshift=-4pt]
    at (ca_col.north) {Common Ancestor $H_{\mathrm{CA}}(m_i,m_j)$};
  \node[anchor=south, font=\tiny, text=BaseInk, yshift=3pt]
    at (ca_col.south) {Co-primary infection $(m_i,m_j)=(0,0)$};

  \node[sampled case] (adi) at ($(ad_col.center)+(0,0.75)$) {$i$};
  \node[latent case] (adm) at ($(ad_col.center)+(0,0.0)$) {$m$};
  \node[sampled case] (adj) at ($(ad_col.center)+(0,-0.75)$) {$j$};
  \draw[scenario flow] (adi) -- (adm);
  \draw[scenario flow] (adm) -- (adj);

  \node[latent case] (caa) at ($(ca_col.center)+(0,0.82)$) {$a$};
  \node[latent case] (cami) at ($(ca_col.center)+(-0.8,0.05)$) {$m_i$};
  \node[latent case] (camj) at ($(ca_col.center)+(0.8,0.05)$) {$m_j$};
  \node[sampled case] (cai) at ($(ca_col.center)+(-0.8,-0.78)$) {$i$};
  \node[sampled case] (caj) at ($(ca_col.center)+(0.8,-0.78)$) {$j$};
  \draw[scenario flow] (caa) -- (cami);
  \draw[scenario flow] (caa) -- (camj);
  \draw[scenario flow] (cami) -- (cai);
  \draw[scenario flow] (camj) -- (caj);

  % --- SHARED MODEL ---
  \node[shared box, text width=4.5cm, anchor=north] (nh_model) at ($(scenarios_border.south)+(0,-0.55)$) {
    \boxheader{Process-Based Model}
    \nodepart{second}
    Gamma stages ($E/P/I$) \\
    \hspace{0.7em} Incubation period: $\tau_{\mathrm{inc}}=y_E+y_P$ \\
    \hspace{0.7em} Generation time: $\tau_{\mathrm{gen}}=y_E+y_{\mathrm{toit}}$ \\
    Observation interval: $\tau_{\mathrm{obs}}=\tau_{\mathrm{inc}}+x_{\mathrm{test}}$ \\
    Clock rate: $\lambda=\nu L/365$ (strict or relaxed)
  };

  % Leave room above both branch boxes for horizontal routing and arrow stems.
  \coordinate (split) at ($(nh_model.south)+(0,-0.35)$);

  % --- TEMPORAL AND GENETIC BRANCHES ---
  \node[temporal box, text width=5.75cm, anchor=north] (t_draws) at ($(split)+(-3.25,-0.45)$) {
    \boxheader{Monte Carlo Time Draws}
    \nodepart{second}
    AD: $T_s=\sum_{r=0}^{m}\tau_{\mathrm{gen},r}+\tau_{\mathrm{obs},j}-\tau_{\mathrm{obs},i}$\\[2pt]
    CA: $T_s=\sum_{s=1}^{m_j+1}\tau_{\mathrm{gen},s}^{(j)}-\sum_{r=1}^{m_i+1}\tau_{\mathrm{gen},r}^{(i)}$\\
    \hspace{1.7em}$+\tau_{\mathrm{obs},j}-\tau_{\mathrm{obs},i}$\\[2pt]
    \textit{Signed difference for temporal ordering}
  };

  \node[genetic box, text width=5.75cm, anchor=north] (g_draws) at ($(split)+(3.25,-0.45)$) {
    \boxheader{Monte Carlo Genetic Draws}
    \nodepart{second}
    AD: $B_s=\max\{0,T_s\}$\\[2pt]
    CA: $B_s=\sum \tau_{\mathrm{gen}}^{(i)}+\sum \tau_{\mathrm{gen}}^{(j)}+\tau_{\mathrm{obs},i}+\tau_{\mathrm{obs},j}$\\[2pt]
    Deterministic: $G_s=\lambda B_s$\\[2pt]
    Stochastic: $G_s\sim\operatorname{Poisson}(\lambda B_s)$
  };

  % Align the compatibility boxes below whichever draw box is taller.
  \path let \p1=(t_draws.south), \p2=(g_draws.south) in
    coordinate (compat_level) at (6.25,{min(\y1,\y2)-0.55cm});
  \node[neutral box, text width=3.75cm, anchor=north] (t_comp) at ($(compat_level)+(-3.25,0)$) {
    \boxheader{Temporal Compatibility}
    \nodepart{second}
    $C_{T,s}=1-2|\Pr(T_s \le t_{ij})-0.5|$
  };

  \node[neutral box, text width=3.75cm, anchor=north] (g_comp) at ($(compat_level)+(3.25,0)$) {
    \boxheader{Genetic Compatibility}
    \nodepart{second}
    $C_{G,s}=1-2|\Pr(G_s \le g_{ij})-0.5|$
  };

  % Combine both scores before the final downward arrow.
  \path let \p1=(t_comp.south), \p2=(g_comp.south) in
    coordinate (merge) at (6.25,{min(\y1,\y2)-0.35cm});
  \node[final box, text width=4.85cm, anchor=north] (final) at ($(merge)+(0,-0.45)$) {
    \boxheader{Aggregated Compatibility Score}
    \nodepart{second}
    Scenario: $C_s(i,j)=C_{T,s}\cdot C_{G,s}$ \\
    Target Set: $C_{\mathrm{S}_\star}(i,j)=\sum_{s\in\mathrm{S}_\star}C_s(i,j)$
  };

  % --- CONNECTIONS ---
  \draw[flow] (input.south) -- (scenarios_border.north);
  \draw[flow] (scenarios_border.south) -- (nh_model.north);

  % Plain lines at junctions; arrowheads only at destination boxes.
  \draw[connector] (nh_model.south) -- (split);
  \draw[flow] (split) -| (t_draws.north);
  \draw[flow] (split) -| (g_draws.north);

  \draw[flow] (t_draws.south) -- (t_comp.north);
  \draw[flow] (g_draws.south) -- (g_comp.north);

  \draw[connector] (t_comp.south) |- (merge);
  \draw[connector] (g_comp.south) |- (merge);
  \draw[flow] (merge) -- (final.north);

\end{tikzpicture}
